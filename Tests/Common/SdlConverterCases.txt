# The SDL rows both converters read alike: x360ce 4's (Engine\Common\Sdl\SdlConverter.cs) and the xinput library's
# (Native\x360ce\Sdl.cpp). SdlNativeTest runs every case through both.
#
# device: the device's axes X to Rz as bits 0 to 5, its sliders as bits 0 and 1, and how many buttons and hats it has;
#         nought buttons or hats lets any through.
# row:    the SDL row.
# expect: every box the row fills, as x360ce 4 writes it, in any order; nothing after "expect:" means no box.
# A case ends at its expect line. A comment line before a case says what it shows.

# The Xbox 360 controller: its triggers share Z, so they are its two halves; Y is read upward; the hat is the D-Pad.
device: axes=0x1F sliders=0x0 buttons=12 povs=1
row: 030000005e0400008e02000000000000,Xbox 360 Controller,a:b0,b:b1,back:b6,dpdown:h0.4,dpleft:h0.8,dpright:h0.2,dpup:h0.1,leftshoulder:b4,leftstick:b8,lefttrigger:+a2,leftx:a0,lefty:a1,rightshoulder:b5,rightstick:b9,righttrigger:-a2,rightx:a3,righty:a4,start:b7,x:b2,y:b3,platform:Windows,
expect: ButtonA=1 ButtonB=2 ButtonBack=7 ButtonStart=8 ButtonX=3 ButtonY=4 DPad=p1 LeftShoulder=5 LeftThumbAxisX=a1 LeftThumbAxisY=a-2 LeftThumbButton=9 LeftTrigger=x3 RightShoulder=6 RightThumbAxisX=a4 RightThumbAxisY=a-5 RightThumbButton=10 RightTrigger=x-3

# An inverted axis: SDL's Y read the other way round is the Xbox controller's way up; a trigger on it is read inverted.
device: axes=0x3F sliders=0x0 buttons=12 povs=1
row: 030000005e0400008e02000000000000,Pad,lefty:a1~,lefttrigger:a2~,righttrigger:a5,platform:Windows,
expect: LeftThumbAxisY=a2 LeftTrigger=a-3 RightTrigger=a6

# SDL counts only the axes a device has: with X, Y and Rz, a2 is Rz and a3 is none.
device: axes=0x23 sliders=0x0 buttons=12 povs=1
row: 03000000300f00000a01000000000000,Pad,a:b0,leftx:a0,lefty:a1,lefttrigger:a2,righttrigger:a3,platform:Windows,
expect: ButtonA=1 LeftThumbAxisX=a1 LeftThumbAxisY=a-2 LeftTrigger=a6

# The sliders come after the axes: with X, Y and one slider, a2 is the slider.
device: axes=0x3 sliders=0x1 buttons=12 povs=1
row: 03000000300f00000a01000000000000,Pad,a:b0,leftx:a0,lefty:a1,lefttrigger:a2,righttrigger:a3,platform:Windows,
expect: ButtonA=1 LeftThumbAxisX=a1 LeftThumbAxisY=a-2 LeftTrigger=s1

# A button or hat the device does not have is left out; so is a half axis read inverted; the first binding of a box wins.
device: axes=0x3 sliders=0x0 buttons=12 povs=1
row: 03000000300f00000a01000000000000,Pad,a:b0,a:b5,b:b20,x:h1.1,y:+a0~,platform:Windows,
expect: ButtonA=1
